33/// Client is the main handle for interacting with iroh-services. It communicates with
34/// iroh-services entirely through an iroh endpoint, and is configured through a builder.
35/// Client requires either an Ssh Key or [`ApiSecret`]
36///
37/// ```no_run
38/// use iroh::{Endpoint, endpoint::presets};
39/// use iroh_services::Client;
40///
41/// async fn build_client() -> anyhow::Result<()> {
42///     let endpoint = Endpoint::bind(presets::N0).await?;
43///
44///     // needs IROH_SERVICES_API_SECRET set to an environment variable
45///     // client will now push endpoint metrics to iroh-services.
46///     let client = Client::builder(&endpoint)
47///         .api_secret_from_str("MY_API_SECRET")?
48///         .build()
49///         .await;
50///
51///     Ok(())
52/// }
53/// ```
54///
55/// [`ApiSecret`]: crate::api_secret::ApiSecret
56#[derive(Debug, Clone)]
57pub struct Client {
58    // owned clone of the endpoint for diagnostics, and for connection restarts on actor close
59    endpoint: Endpoint,
60    message_channel: tokio::sync::mpsc::Sender<ClientActorMessage>,
61    _actor_task: Arc<AbortOnDropHandle<()>>,
62}

118    /// Set an optional human-readable name for the endpoint, making its metrics
119    /// easier to identify.
120    ///
121    /// Often a database user id, machine name, or other stable identifier from
122    /// your application. A name must be 2 to 128 bytes of UTF-8; uniqueness is not
123    /// enforced, so different endpoints may share a name.
124    ///
125    /// Validation errors are returned here. The name is sent to the server after
126    /// the client authenticates; a failure to send it at that point is logged at
127    /// warn level rather than returned; use [`Client::set_name`] to set it later
128    /// with explicit error handling.
129    pub fn name(mut self, name: impl Into<String>) -> Result<Self> {
130        let name = name.into();
131        validate_name(&name).map_err(BuildError::InvalidName)?;
132        self.name = Some(name);
133        Ok(self)
134    }

136    /// Attach the endpoint to a single named group when the client first
137    /// authenticates.
138    ///
139    /// A group name must be 2 to 128 bytes of UTF-8. Validation errors are returned
140    /// here. The group is sent to the server after the client authenticates; a
141    /// failure to send it at that point is logged at warn level rather than
142    /// returned; use [`Client::set_group`] to set it later with explicit error

151    /// Attach arbitrary key-value attributes to the endpoint when the client
152    /// first authenticates. Accepts any iterable of `(key, value)` pairs:
153    ///
154    /// ```no_run
155    /// # use iroh::{Endpoint, endpoint::presets};
156    /// # use iroh_services::Client;
157    /// # async fn example(endpoint: &Endpoint) -> anyhow::Result<()> {
158    /// let _ = Client::builder(endpoint).attributes([("env", "prod"), ("region", "us-west")])?;
159    /// # Ok(()) }
160    /// ```
161    ///
162    /// Each key must be 2 to 128 bytes of UTF-8; values may be empty and are capped
163    /// at 128 bytes; at most 128 entries are allowed. Validation errors are
164    /// returned here. The attributes are sent to the server after the client
165    /// authenticates; a failure to send them at that point is logged at warn
166    /// level rather than returned; use [`Client::set_attributes`] to set them
167    /// later with explicit error handling.
168    pub fn attributes<I, K, V>(mut self, attrs: I) -> Result<Self>
169    where
170        I: IntoIterator<Item = (K, V)>,
171        K: Into<String>,
172        V: Into<String>,
173    {
174        let collected: BTreeMap<String, String> = attrs
175            .into_iter()
176            .map(|(k, v)| (k.into(), v.into()))
177            .collect();
178        validate_attributes(&collected).map_err(BuildError::InvalidAttributes)?;
179        self.attributes = Some(collected);
180        Ok(self)
181    }

183    /// Check IROH_SERVICES_API_SECRET environment variable for a valid API secret
184    pub fn api_secret_from_env(self) -> Result<Self> {

189    /// set client API secret from an encoded string
190    pub fn api_secret_from_str(self, secret_key: &str) -> Result<Self> {
191        let key = ApiSecret::from_str(secret_key).context("invalid iroh services api secret")?;
192        self.api_secret(key)

195    /// Use a shared secret & remote iroh-services endpoint ID contained within a ticket
196    /// to construct a iroh-services client. The resulting client will have "Client"
197    /// capabilities.
198    ///
199    /// API secrets include remote details within them, and will set both the
200    /// remote and rcan values on the builder
201    pub fn api_secret(mut self, ticket: ApiSecret) -> Result<Self> {
202        let local_id = self.endpoint.id();
203        let rcan = crate::caps::create_api_token_from_secret_key(
204            ticket.secret,
205            local_id,
206            self.cap_expiry,
207            Caps::for_shared_secret(),
208        )?;
209
210        self.remote = Some(ticket.remote);
211        self.rcan(rcan)
212    }

214    /// Loads the private ssh key from the given path, and creates the needed capability.
215    ///
216    /// The file must contain an unencrypted PEM-encoded OpenSSH ed25519 private key.
217    #[cfg(not(wasm_browser))]
218    pub async fn ssh_key_from_file<P: AsRef<std::path::Path>>(self, path: P) -> Result<Self> {
219        let file_content = tokio::fs::read_to_string(path).await?;
220        self.ssh_key(&file_content)
221    }

223    /// Creates the capability from the provided PEM-encoded OpenSSH ed25519 private key.
224    #[cfg(not(wasm_browser))]
225    pub fn ssh_key(mut self, pem: &str) -> Result<Self> {
226        let local_id = self.endpoint.id();
227        let rcan = crate::caps::create_api_token_from_openssh_pem(
228            pem,
229            local_id,
230            self.cap_expiry,
231            Caps::all(),
232        )?;
233        self.cap.replace(rcan);
234
235        Ok(self)
236    }

238    /// Sets the rcan directly.
239    pub fn rcan(mut self, cap: Rcan<Caps>) -> Result<Self> {
240        ensure!(
241            EndpointId::from_verifying_key(*cap.audience()) == self.endpoint.id(),
242            "invalid audience"
243        );
244        self.cap.replace(cap);
245        Ok(self)
246    }

255    /// Create a new client, connected to the provide service node
256    #[must_use = "dropping the client will silently cancel all client tasks"]
257    pub async fn build(self) -> Result<Client, BuildError> {
258        debug!("starting iroh-services client");
259        let remote = self.remote.ok_or(BuildError::MissingRemote)?;
260        let capabilities = self.cap.ok_or(BuildError::MissingCapability)?;
261
262        let registry = Arc::new(RwLock::new(self.registry));
263        let (tx, rx) = tokio::sync::mpsc::channel(1);
264        let actor_task = AbortOnDropHandle::new(n0_future::task::spawn(
265            ClientActor {
266                capabilities,
267                endpoint: self.endpoint.clone(),
268                remote,
269                client: None,
270                name: self.name.clone(),
271                group: self.group.clone(),
272                attributes: self.attributes.clone().unwrap_or_default(),
273                session_id: Uuid::new_v4(),
274                encoder: Encoder::new(registry.clone()),
275                registry,
276            }
277            .run(self.metrics_interval, rx),
278        ));
279
280        Ok(Client {
281            endpoint: self.endpoint,
282            message_channel: tx,
283            _actor_task: Arc::new(actor_task),
284        })
285    }

288#[derive(thiserror::Error, Debug)]
289#[non_exhaustive]
290pub enum BuildError {
291    #[error("Missing remote endpoint to dial")]
292    MissingRemote,
293    #[error("Missing capability")]
294    MissingCapability,
295    #[error("Unauthorized")]
296    Unauthorized,
297    #[error("Remote error: {0}")]
298    Remote(#[from] RemoteError),
299    #[error("Rpc connection error: {0}")]
300    Rpc(irpc::Error),
301    #[error("Connection error: {0}")]
302    Connect(ConnectError),
303    #[error("Invalid endpoint name: {0}")]
304    InvalidName(#[from] ValidateNameError),
305    #[error("Invalid endpoint group: {0}")]
306    InvalidGroup(ValidateNameError),
307    #[error("Invalid endpoint attributes: {0}")]
308    InvalidAttributes(#[from] ValidateAttributesError),
309}
310
311impl From<irpc::Error> for BuildError {
312    fn from(value: irpc::Error) -> Self {
313        match value {
314            irpc::Error::Request {
315                source:
316                    irpc::RequestError::Connection {
317                        source: iroh::endpoint::ConnectionError::ApplicationClosed(frame),
318                        ..
319                    },
320                ..
321            } if frame.error_code == 401u32.into() => Self::Unauthorized,
322            value => Self::Rpc(value),
323        }
324    }

327/// Minimum length in bytes for an endpoint name.
328pub const CLIENT_NAME_MIN_LENGTH: usize = 2;
329/// Maximum length in bytes for an endpoint name.
330pub const CLIENT_NAME_MAX_LENGTH: usize = 128;
331
332/// Error returned when an endpoint name fails validation.
333#[derive(Debug, thiserror::Error)]
334pub enum ValidateNameError {
335    #[error("Name is too long (must be no more than {CLIENT_NAME_MAX_LENGTH} bytes).")]
336    TooLong,
337    #[error("Name is too short (must be at least {CLIENT_NAME_MIN_LENGTH} bytes).")]
338    TooShort,
339}
340
341fn validate_name(name: &str) -> Result<(), ValidateNameError> {
342    if name.len() < CLIENT_NAME_MIN_LENGTH {
343        Err(ValidateNameError::TooShort)
344    } else if name.len() > CLIENT_NAME_MAX_LENGTH {
345        Err(ValidateNameError::TooLong)
346    } else {
347        Ok(())
348    }
349}
350
351/// Maximum length in bytes for an attribute value. Values may be empty.
352pub const CLIENT_ATTRIBUTE_VALUE_MAX_LENGTH: usize = 128;
353/// Maximum number of entries allowed in the attributes map.
354pub const CLIENT_ATTRIBUTES_MAX_COUNT: usize = 128;
355
356/// Error returned when an attributes map fails validation.
357#[derive(Debug, thiserror::Error)]
358pub enum ValidateAttributesError {
359    #[error("Too many attributes (must be no more than {CLIENT_ATTRIBUTES_MAX_COUNT}).")]
360    TooManyEntries,
361    #[error("Invalid attribute key: {0}")]
362    InvalidKey(#[from] ValidateNameError),
363    #[error(
364        "Attribute value too long (must be no more than {CLIENT_ATTRIBUTE_VALUE_MAX_LENGTH} bytes)."
365    )]
366    ValueTooLong,
367}
368
369fn validate_attributes(attrs: &BTreeMap<String, String>) -> Result<(), ValidateAttributesError> {
370    if attrs.len() > CLIENT_ATTRIBUTES_MAX_COUNT {
371        return Err(ValidateAttributesError::TooManyEntries);
372    }
373    for (k, v) in attrs {
374        validate_name(k)?;
375        if v.len() > CLIENT_ATTRIBUTE_VALUE_MAX_LENGTH {
376            return Err(ValidateAttributesError::ValueTooLong);
377        }
378    }
379    Ok(())
380}

385    #[error("Invalid endpoint name: {0}")]
386    InvalidName(#[from] ValidateNameError),

389    #[error("Invalid endpoint attributes: {0}")]

438    /// Name the active endpoint cloud-side.
439    ///
440    /// names can be any UTF-8 string, with a min length of 2 bytes, and
441    /// maximum length of 128 bytes. **name uniqueness is not enforced.**
442    pub async fn set_name(&self, name: impl Into<String>) -> Result<(), Error> {

445        debug!(name_len = name.len(), "calling set name");
446        let (tx, rx) = oneshot::channel();

453    /// Attach the active endpoint to a single named group cloud-side.
454    ///
455    /// A group name must be 2 to 128 bytes of UTF-8.

459        debug!(%group, "calling set group");

467    /// Replace the arbitrary key-value attributes on the active endpoint cloud-side.
468    ///
469    /// Accepts any iterable of `(key, value)` pairs (arrays of tuples, `Vec`s,
470    /// `HashMap`s, `BTreeMap`s, etc.), so most calls fit on a single line:
471    ///
472    /// ```no_run
473    /// # use iroh_services::Client;
474    /// # async fn example(client: Client) -> anyhow::Result<()> {
475    /// client
476    ///     .set_attributes([("env", "prod"), ("region", "us-west")])
477    ///     .await?;
478    /// # Ok(()) }
479    /// ```
480    ///
481    /// Each key must be 2 to 128 bytes of UTF-8; values may be empty and are limited
482    /// to 128 bytes; at most 128 entries are allowed. Each call fully replaces
483    /// the prior set; passing an empty iterator clears all attributes.
484    pub async fn set_attributes<I, K, V>(&self, attrs: I) -> Result<(), Error>
485    where
486        I: IntoIterator<Item = (K, V)>,
487        K: Into<String>,
488        V: Into<String>,
489    {
490        let collected: BTreeMap<String, String> = attrs
491            .into_iter()
492            .map(|(k, v)| (k.into(), v.into()))
493            .collect();
494        validate_attributes(&collected)?;
495        debug!(attr_count = collected.len(), "calling set attributes");
496        let (tx, rx) = oneshot::channel();
497        self.message_channel
498            .send(ClientActorMessage::SetAttributes {
499                attributes: collected,
500                done: tx,
501            })
502            .await?;
503        rx.await?
504    }
505
506    /// Set or replace a single attribute, merging it into the endpoint's existing
507    /// attributes rather than replacing the whole set.
508    ///
509    /// A convenience over [`set_attributes`](Self::set_attributes) when you only
510    /// need to change one value. The key must be 2 to 128 bytes of UTF-8 and the
511    /// value is limited to 128 bytes; the merged set must stay within 128 entries.
512    pub async fn set_attribute(
513        &self,
514        key: impl Into<String>,
515        value: impl Into<String>,
516    ) -> Result<(), Error> {
517        // Validation happens in the actor against the merged set (current
518        // attributes plus this entry), since only there is the current set
519        // known. Merging can exceed the entry-count limit even when this
520        // single entry is valid.
521        let (tx, rx) = oneshot::channel();
522        self.message_channel
523            .send(ClientActorMessage::SetAttribute {
524                key: key.into(),
525                value: value.into(),
526                done: tx,
527            })
528            .await?;
529        rx.await?
530    }

552    /// Grant capabilities to a remote endpoint. Creates a signed RCAN token
553    /// and sends it to iroh-services for storage. The remote can then use this token

572                done: tx,

574            .await?;

578    /// run local network status diagnostics, optionally uploading the results
579    pub async fn net_diagnostics(&self, send: bool) -> Result<DiagnosticsReport, Error> {
580        let report = run_diagnostics(&self.endpoint).await?;
581        if send {

583            self.message_channel

640/// An irpc client bound to a single authenticated connection.
641///
642/// The server accepts requests only after an `Auth` as the very first
643/// request on a connection, so [`RpcClient::connect`] dials and
644/// authenticates in one step; a value of this type never exists
645/// unauthenticated.
646struct RpcClient {
647    /// Kept alongside the irpc client to detect a remote close.
648    connection: Connection,
649    irpc: IrohServicesClient,
650}
651
652impl RpcClient {
653    /// Dials the remote and authenticates as the connection's first request.
654    async fn connect(
655        endpoint: &Endpoint,
656        remote: EndpointAddr,
657        caps: Rcan<Caps>,
658    ) -> Result<Self, Error> {
659        trace!("client connecting and authorizing");
660        let connection = endpoint
661            .connect(remote, ALPN)
662            .await
663            .inspect_err(|err| debug!("connect failed: {err:?}"))?;
664        let irpc = IrohServicesClient::boxed(IrohRemoteConnection::new(connection.clone()));
665        irpc.rpc(Auth { caps })
666            .await
667            .inspect_err(|err| debug!("authorization failed: {err:?}"))
668            .map_err(|err| RemoteError::AuthError(err.to_string()))?;
669        Ok(Self { connection, irpc })
670    }

673struct ClientActor {
674    capabilities: Rcan<Caps>,
675    endpoint: Endpoint,
676    remote: EndpointAddr,
677    /// The active authenticated connection, established on demand.
678    ///
679    /// The actor owns the connection lifecycle rather than using a lazy
680    /// reconnecting client: the server requires `Auth` as the first request
681    /// on every connection, so a transparent mid-request reconnect would send
682    /// an unauthenticated request that the server rejects. [`Self::connect`]
683    /// establishes this; [`Self::rpc`] clears it on transport errors so the
684    /// next request re-dials and re-authenticates.
685    client: Option<RpcClient>,
686    name: Option<String>,
687    group: Option<String>,
688    attributes: BTreeMap<String, String>,
689    session_id: Uuid,
690    encoder: Encoder,
691    /// Kept so connect() can rebuild the encoder to re-send the metrics schema.
692    registry: Arc<RwLock<Registry>>,
693}
694
695impl ClientActor {
696    async fn run(
697        mut self,
698        interval: Option<Duration>,
699        mut inbox: tokio::sync::mpsc::Receiver<ClientActorMessage>,
700    ) {
701        let mut metrics_timer = interval.map(|interval| n0_future::time::interval(interval));
702        trace!("starting client actor");
703
704        // Send the initial metadata (set via the builder) once the actor starts.
705        // These live on `self`; a send failure here is logged, not fatal.
706        if let Some(name) = self.name.clone()
707            && let Err(err) = self.send_name_endpoint(name).await
708        {
709            warn!(err = %err, "failed setting endpoint name on startup");
710        }
711
712        if let Some(group) = self.group.clone()
713            && let Err(err) = self.send_set_group(group).await
714        {
715            warn!(err = %err, "failed setting endpoint group on startup");
716        }
717
718        if !self.attributes.is_empty()
719            && let Err(err) = self.send_set_attributes(self.attributes.clone()).await
720        {
721            warn!(err = %err, "failed setting endpoint attributes on startup");
722        }
723
724        loop {
725            trace!("client actor tick");
726            tokio::select! {
727                biased;
728                Some(msg) = inbox.recv() => {
729                    match msg {
730                        ClientActorMessage::Ping { done } => {
731                            let res = self.send_ping().await;
732                            done.send(res).ok();
733                        },
734                        ClientActorMessage::SendMetrics { done } => {
735                            trace!("sending metrics manually triggered");
736                            let res = self.send_metrics().await;
737                            done.send(res).ok();
738                        }
739                        ClientActorMessage::GrantCap { cap, done } => {
740                            let res = self.grant_cap(*cap).await;
741                            done.send(res).ok();
742                        }
743                        ClientActorMessage::ReadName { done } => {
744                            done.send(self.name.clone()).ok();
745                        }
746                        ClientActorMessage::ReadGroup { done } => {
747                            done.send(self.group.clone()).ok();
748                        }
749                        ClientActorMessage::NameEndpoint { name, done } => {
750                            let res = self.send_name_endpoint(name).await;
751                            done.send(res).ok();
752                        }
753                        ClientActorMessage::SetGroup { group, done } => {
754                            let res = self.send_set_group(group).await;
755                            done.send(res).ok();
756                        }
757                        ClientActorMessage::SetAttributes { attributes, done } => {
758                            let res = self.send_set_attributes(attributes).await;
759                            done.send(res).ok();
760                        }
761                        ClientActorMessage::SetAttribute { key, value, done } => {
762                            // Merge into the current set and validate the union:
763                            // adding one valid entry to a valid set can still
764                            // exceed the max entry count, so the single entry
765                            // being valid is not enough.
766                            let mut merged = self.attributes.clone();
767                            merged.insert(key, value);
768                            let res = match validate_attributes(&merged) {
769                                Ok(()) => self.send_set_attributes(merged).await,
770                                Err(err) => Err(Error::from(err)),
771                            };
772                            done.send(res).ok();
773                        }
774                        ClientActorMessage::PutNetworkDiagnostics { report, done } => {
775                            let res = self.put_network_diagnostics(*report).await;
776                            done.send(res).ok();
777                        }
778                    }
779                }
780                _ = async {
781                    if let Some(ref mut timer) = metrics_timer {
782                        timer.tick().await;
783                    } else {
784                        std::future::pending::<()>().await;
785                    }
786                } => {
787                    trace!("metrics send tick");
788                    if let Err(err) = self.send_metrics().await {
789                        debug!("failed to push metrics: {:#?}", err);
790                    }
791                },
792            }
793        }
794    }
795
796    /// Returns the client for the active connection, establishing one if needed.
797    ///
798    /// The server keeps one metrics decoder per connection, so a fresh
799    /// connection also gets a fresh encoder: the next metrics export then
800    /// carries the full schema for the server's fresh decoder.
801    async fn connect(&mut self) -> Result<&IrohServicesClient, Error> {
802        // A connection the remote has closed (server restart, error close)
803        // can't carry requests anymore; drop it and re-dial.
804        if let Some(client) = &self.client
805            && let Some(reason) = client.connection.close_reason()
806        {
807            debug!(%reason, "connection closed by remote, reconnecting");
808            self.client = None;
809        }
810        if self.client.is_none() {
811            let client = RpcClient::connect(
812                &self.endpoint,
813                self.remote.clone(),
814                self.capabilities.clone(),
815            )
816            .await?;
817            // Recreate the metrics encoder to force sending the schema on the next metrics push.
818            self.encoder = Encoder::new(self.registry.clone());
819            self.client = Some(client);
820        }
821        Ok(&self.client.as_ref().expect("just checked").irpc)
822    }
823
824    async fn rpc<Req, Res>(&mut self, msg: Req) -> Result<Res, Error>
825    where
826        IrohServicesProtocol: From<Req>,
827        ServicesMessage: From<WithChannels<Req, IrohServicesProtocol>>,
828        Req: Channels<
829                IrohServicesProtocol,
830                Tx = irpc::channel::oneshot::Sender<Res>,
831                Rx = NoReceiver,
832            > + Display,
833        Res: RpcMessage,
834    {
835        trace!(request = %msg, "client actor send request");
836        let client = self.connect().await?;
837        let res = client.rpc(msg).await;
838        if res.is_err() {
839            // A transport-level failure means the connection is gone or in an
840            // unknown state; the next request dials and authenticates afresh.
841            // Server-side errors arrive as `Ok(Err(RemoteError))` and keep the
842            // connection.
843            self.client = None;
844        }
845        res.inspect_err(|err| warn!("rpc error: {err}"))
846            .map_err(Error::from)
847    }

851        self.rpc(Ping { req_id: req }).await
852    }

894    }

952    /// In-process stand-in for the services backend.
953    ///
954    /// Mirrors the real server's session rules: the first request on every
955    /// connection must be Auth, a repeat Auth on a live connection closes it,
956    /// and the metrics decoder lives per connection.

958    struct RecordingServer {
959        seen: tokio::sync::mpsc::UnboundedSender<SeenUpdate>,
960        drop_next: Arc<AtomicBool>,

963    impl ProtocolHandler for RecordingServer {
964        async fn accept(&self, connection: Connection) -> Result<(), AcceptError> {
965            self.handle_connection(connection).await.map_err(|e| {
966                let boxed: Box<dyn std::error::Error + Send + Sync> = e.into();
967                AcceptError::from(AnyError::from(boxed))
968            })
969        }

972    impl RecordingServer {
973        async fn handle_connection(&self, connection: Connection) -> anyhow::Result<()> {
974            let Some(first_request) = read_request::<IrohServicesProtocol>(&connection).await?
975            else {
976                return Ok(());
977            };
978            let ServicesMessage::Auth(WithChannels { tx, .. }) = first_request else {
979                connection.close(400u32.into(), b"Expected initial auth message");
980                return Ok(());
981            };
982            tx.send(()).await?;
983
984            let mut decoder = Decoder::default();
985            loop {
986                let Some(request) = read_request::<IrohServicesProtocol>(&connection).await? else {
987                    return Ok(());
988                };
989                if self.drop_next.swap(false, Ordering::SeqCst) {
990                    // Simulates a server restart: the connection dies without an
991                    // answer and takes the per-connection decoder with it.
992                    connection.close(500u32.into(), b"test restart");
993                    return Ok(());
994                }
995                match request {
996                    ServicesMessage::Auth(_) => {
997                        connection.close(400u32.into(), b"Unexpected auth message");
998                        anyhow::bail!("client re-sent auth on a live connection");
999                    }
1000                    ServicesMessage::PutMetrics(WithChannels { inner, tx, .. }) => {
1001                        let has_schema = inner.update.schema.is_some();
1002                        decoder.import(inner.update);
1003                        let decoded_items = decoder.iter().count();
1004                        let _ = self.seen.send(SeenUpdate {
1005                            has_schema,
1006                            decoded_items,
1007                        });
1008                        tx.send(Ok(())).await?;
1009                    }
1010                    _ => {
1011                        connection.close(400u32.into(), b"Unexpected message in test");
1012                        anyhow::bail!("unexpected message in test");
1013                    }
1014                }
1015            }
1016        }

1030        let drop_next = Arc::new(AtomicBool::new(false));
1031        let server = RecordingServer {
1032            seen: seen_tx,
1033            drop_next: drop_next.clone(),
1034        };
1035        let router = Router::builder(server_ep.clone())
1036            .accept(ALPN, server)
1037            .spawn();
1038
1039        // The test server accepts any capability, so a self-issued token works.
1040        let shared_secret = SecretKey::from_bytes(&rng.random());
1041        let cap = create_api_token_from_secret_key(
1042            shared_secret,
1043            client_ep.id(),
1044            Duration::from_secs(3600),
1045            Caps::for_shared_secret(),
1046        )
1047        .unwrap();

1049        let client = Client::builder(&client_ep)
1050            .disable_metrics_interval()
1051            .remote(server_ep.addr())
1052            .rcan(cap)
1053            .unwrap()
1054            .build()
1055            .await
1056            .unwrap();

1059        client.push_metrics().await.unwrap();
1060        let first = seen_rx.recv().await.unwrap();
1061        assert!(first.has_schema);
1062        assert!(first.decoded_items > 0);
1063
1064        // Steady state stops sending the schema. The endpoint's own metric
1065        // families may still grow for a few exports right after connecting
1066        // (each growth re-publishes the schema), so push until the schema
1067        // settles; the connection's decoder keeps decoding throughout.
1068        let mut settled = false;
1069        for _ in 0..20 {
1070            client.push_metrics().await.unwrap();
1071            let seen = seen_rx.recv().await.unwrap();
1072            assert!(seen.decoded_items > 0);
1073            if !seen.has_schema {
1074                settled = true;
1075                break;
1076            }
1077        }
1078        assert!(settled, "schema must stop being sent once it is unchanged");
1079
1080        // The server drops the connection mid-request: one failed round trip.
1081        drop_next.store(true, Ordering::SeqCst);
1082        assert!(client.push_metrics().await.is_err());
1083
1084        // The client re-dials and re-auths; the first update on the new
1085        // connection must include the schema for the server's fresh decoder.
1086        client.push_metrics().await.unwrap();
1087        let third = seen_rx.recv().await.unwrap();
1088        assert!(third.has_schema, "schema must be re-sent after a reconnect");
1089        assert!(third.decoded_items > 0);
1090
1091        router.shutdown().await.unwrap();
1092        client_ep.close().await;
1093    }

1312    /// Build a client with no reachable server, mirroring `test_no_metrics_interval`.
1313    /// The runtime setters validate input locally before any network call, so
1314    /// validation errors surface without a live server.
1315    async fn build_serverless_client(seed: u64) -> Client {
1316        let mut rng = rand::rngs::ChaCha8Rng::seed_from_u64(seed);
1317        let shared_secret = SecretKey::from_bytes(&rng.random());
1318        let fake_endpoint_id = SecretKey::from_bytes(&rng.random()).public();
1319        let api_secret = ApiSecret::new(shared_secret, fake_endpoint_id);
1320
1321        let endpoint = Endpoint::builder(presets::Minimal).bind().await.unwrap();
1322
1323        Client::builder(&endpoint)
1324            .disable_metrics_interval()
1325            .api_secret(api_secret)
1326            .unwrap()
1327            .build()
1328            .await
1329            .unwrap()
1330    }
